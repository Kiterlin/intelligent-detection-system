## 🔧 技术实现要点

### Celery Group使用示例

```python
from celery import group

# 创建任务组
job = group(
    process_single_file_detection.s(file.id, model_id, params, batch_job_id)
    for file in dataset.files
)

# 执行任务组
result = job.apply_async()

# 保存group_id
batch_job.celery_group_id = result.id
```

### 进度追踪机制

```python
# 在每个单文件任务完成时
batch_job.completed_files += 1
if detection_failed:
    batch_job.failed_files += 1

# 检查是否全部完成
if batch_job.completed_files + batch_job.failed_files == batch_job.total_files:
    batch_job.status = BatchJobStatus.COMPLETED
    batch_job.completed_at = datetime.now()
```

### 前端轮询进度

```javascript
// 启动批量检测后，每3秒轮询一次进度
const pollJobProgress = async (jobId) => {
  const intervalId = setInterval(async () => {
    const response = await getBatchJob(jobId)
    const job = response.data

    // 更新进度
    progress.value = (job.completed_files / job.total_files) * 100

    // 如果完成或失败，停止轮询
    if (job.status === 'completed' || job.status === 'failed') {
      clearInterval(intervalId)
    }
  }, 3000)
}
```

### 文件上传处理

```python
# 后端文件保存
DATASET_FILES_DIR = settings.DATA_DIR / "datasets"

async def save_uploaded_file(file: UploadFile, dataset_id: int):
    dataset_dir = DATASET_FILES_DIR / str(dataset_id)
    dataset_dir.mkdir(parents=True, exist_ok=True)

    file_path = dataset_dir / file.filename
    content = await file.read()

    with open(file_path, "wb") as f:
        f.write(content)

    return str(file_path)
```
