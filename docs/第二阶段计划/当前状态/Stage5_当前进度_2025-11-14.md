#### 4. 开始实现 Celery 任务

**文件**: `backend/app/tasks/report_tasks.py`

**参考代码结构**:
```python
from celery import Task
from app.core.celery_app import celery_app
from app.core.database import async_session_maker
from app.services.report_generator import ReportGenerator
from app.crud import report as report_crud

@celery_app.task(bind=True, queue='reports')
async def generate_report_task(self, report_id: int):
    """异步生成报告任务"""
    async with async_session_maker() as db:
        try:
            # 1. 更新状态为 generating
            await report_crud.update_report_status(
                db, report_id, 'generating'
            )

            # 2. 获取报告信息
            report = await report_crud.get_report(db, report_id)

            # 3. 生成报告
            generator = ReportGenerator()
            pdf_path, html_path = await generator.generate_report(
                db=db,
                report_id=report.id,
                report_type=report.report_type,
                user_id=report.user_id,
                parameters=report.parameters_json
            )

            # 4. 更新状态为 completed
            await report_crud.update_report_status(
                db=db,
                report_id=report_id,
                status='completed',
                file_path_pdf=pdf_path,
                file_path_html=html_path,
                completed_at=datetime.now()
            )

        except Exception as e:
            # 5. 更新状态为 failed
            await report_crud.update_report_status(
                db=db,
                report_id=report_id,
                status='failed',
                error_message=str(e)
            )
            raise
```
