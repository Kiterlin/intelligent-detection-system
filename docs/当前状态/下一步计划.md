from sqlalchemy import Column, Integer, String, DateTime, ForeignKey, Text, Enum
from sqlalchemy.orm import relationship
from datetime import datetime
import enum

class DetectionStatus(str, enum.Enum):
    PENDING = "pending"      # 等待处理
    PROCESSING = "processing"  # 处理中
    COMPLETED = "completed"    # 已完成
    FAILED = "failed"          # 失败

class InputType(str, enum.Enum):
    IMAGE = "image"
    VIDEO = "video"

class DetectionHistory(Base):
    __tablename__ = "detection_history"

    id = Column(Integer, primary_key=True, index=True)
    user_id = Column(Integer, ForeignKey("users.id"), nullable=False)
    model_id = Column(Integer, ForeignKey("ml_models.id"), nullable=False)

    input_type = Column(Enum(InputType), nullable=False)
    input_file_path = Column(String(500), nullable=False)
    output_file_path = Column(String(500), nullable=True)

    result_json = Column(Text, nullable=True)  # 检测结果JSON
    status = Column(Enum(DetectionStatus), default=DetectionStatus.PENDING)

    task_id = Column(String(100), nullable=True)  # Celery任务ID
    error_message = Column(Text, nullable=True)

    detected_objects_count = Column(Integer, default=0)
    processing_time = Column(Float, nullable=True)  # 处理时长（秒）

    created_at = Column(DateTime, default=datetime.utcnow)
    completed_at = Column(DateTime, nullable=True)

    # 关系
    user = relationship("User", foreign_keys=[user_id])
    model = relationship("MLModel", foreign_keys=[model_id])

from ultralytics import YOLO
from pathlib import Path
from typing import Optional
import logging

logger = logging.getLogger(__name__)

class ModelLoader:
    """YOLO模型加载和管理"""

    def __init__(self):
        self._current_model: Optional[YOLO] = None
        self._current_model_id: Optional[int] = None

    async def load_active_model(self, db: AsyncSession) -> YOLO:
        """加载当前激活的模型"""
        # 从数据库获取激活的模型
        active_model = await crud_ml_model.get_active_model(db)

        if not active_model:
            raise ValueError("没有激活的模型")

        # 检查是否需要重新加载
        if self._current_model_id == active_model.id:
            return self._current_model

        # 加载新模型
        model_path = Path(active_model.file_path)
        if not model_path.exists():
            raise FileNotFoundError(f"模型文件不存在: {model_path}")

        logger.info(f"加载模型: {active_model.name} ({active_model.file_path})")
        self._current_model = YOLO(str(model_path))
        self._current_model_id = active_model.id

        return self._current_model

    def get_model(self) -> Optional[YOLO]:
        """获取当前加载的模型"""
        return self._current_model

# 全局单例
model_loader = ModelLoader()

**文件**: `backend/app/services/detection.py`

```python
import cv2
import numpy as np
from PIL import Image
from pathlib import Path
from typing import List, Dict, Tuple
import json

class DetectionService:
    """目标检测服务"""

    @staticmethod
    def detect_image(
        model,
        image_path: str,
        conf_threshold: float = 0.25
    ) -> Dict:
        """
        检测图片

        Returns:
            {
                "objects": [...],
                "count": 5,
                "processing_time": 0.123
            }
        """
        results = model.predict(
            source=image_path,
            conf=conf_threshold,
            save=False
        )

        # 解析结果
        detections = []
        for result in results:
            boxes = result.boxes
            for box in boxes:
                detections.append({
                    "class": result.names[int(box.cls)],
                    "confidence": float(box.conf),
                    "bbox": box.xyxy[0].tolist()  # [x1, y1, x2, y2]
                })

        return {
            "objects": detections,
            "count": len(detections),
            "processing_time": results[0].speed['inference'] / 1000
        }

    @staticmethod
    def annotate_image(
        image_path: str,
        output_path: str,
        detections: List[Dict]
    ):
        """在图片上绘制检测框"""
        image = cv2.imread(image_path)

        for det in detections:
            x1, y1, x2, y2 = map(int, det['bbox'])
            label = f"{det['class']} {det['confidence']:.2f}"

            # 绘制框
            cv2.rectangle(image, (x1, y1), (x2, y2), (0, 255, 0), 2)

            # 绘制标签
            cv2.putText(
                image, label, (x1, y1 - 10),
                cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 2
            )

        cv2.imwrite(output_path, image)

detection_service = DetectionService()
```

```python
from fastapi import APIRouter, UploadFile, File, Depends, HTTPException
from sqlalchemy.ext.asyncio import AsyncSession
from app.api.deps import get_current_user, get_db
from app.services.model_loader import model_loader
from app.services.detection import detection_service
from app.models.user import User
from app.schemas.detection import DetectionResponse
import aiofiles
from pathlib import Path
import uuid

router = APIRouter()

@router.post("/image", response_model=DetectionResponse)
async def detect_image(
    file: UploadFile = File(...),
    conf_threshold: float = 0.25,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db)
):
    """
    图片目标检测

    - **file**: 图片文件 (jpg, png, jpeg)
    - **conf_threshold**: 置信度阈值 (0-1)
    """
    # 验证文件格式
    if not file.filename.lower().endswith(('.jpg', '.jpeg', '.png')):
        raise HTTPException(400, "仅支持 JPG/PNG 格式")

    # 保存上传的图片
    upload_dir = Path("uploads/detections")
    upload_dir.mkdir(parents=True, exist_ok=True)

    file_id = str(uuid.uuid4())
    input_path = upload_dir / f"{file_id}_input{Path(file.filename).suffix}"
    output_path = upload_dir / f"{file_id}_output.jpg"

    async with aiofiles.open(input_path, 'wb') as f:
        content = await file.read()
        await f.write(content)

    try:
        # 加载模型
        model = await model_loader.load_active_model(db)
        active_model = await crud_ml_model.get_active_model(db)

        # 执行检测
        result = detection_service.detect_image(
            model,
            str(input_path),
            conf_threshold
        )

        # 生成标注图片
        detection_service.annotate_image(
            str(input_path),
            str(output_path),
            result['objects']
        )

        # 保存检测记录
        history = DetectionHistory(
            user_id=current_user.id,
            model_id=active_model.id,
            input_type=InputType.IMAGE,
            input_file_path=str(input_path),
            output_file_path=str(output_path),
            result_json=json.dumps(result['objects']),
            status=DetectionStatus.COMPLETED,
            detected_objects_count=result['count'],
            processing_time=result['processing_time']
        )
        db.add(history)
        await db.commit()

        return {
            "detection_id": history.id,
            "objects": result['objects'],
            "count": result['count'],
            "input_url": f"/uploads/detections/{input_path.name}",
            "output_url": f"/uploads/detections/{output_path.name}",
            "processing_time": result['processing_time']
        }

    except Exception as e:
        logger.error(f"检测失败: {e}")
        raise HTTPException(500, f"检测失败: {str(e)}")
```

### Stage 5: 视频检测（Celery异步任务）

#### 5.1 创建Celery任务

**文件**: `backend/app/tasks/detection_tasks.py`

```python
from app.worker import celery_app
from app.services.detection import detection_service
from app.core.database import AsyncSessionLocal
from app.models.detection_history import DetectionHistory, DetectionStatus
import cv2
from pathlib import Path
import logging

logger = logging.getLogger(__name__)

@celery_app.task(bind=True)
def detect_video_task(
    self,
    detection_id: int,
    input_path: str,
    model_path: str,
    conf_threshold: float = 0.25
):
    """
    视频检测异步任务

    Args:
        detection_id: 检测记录ID
        input_path: 输入视频路径
        model_path: 模型路径
        conf_threshold: 置信度阈值
    """
    try:
        # 更新状态为处理中
        async with AsyncSessionLocal() as db:
            history = await db.get(DetectionHistory, detection_id)
            history.status = DetectionStatus.PROCESSING
            history.task_id = self.request.id
            await db.commit()

        # 加载模型
        from ultralytics import YOLO
        model = YOLO(model_path)

        # 处理视频
        cap = cv2.VideoCapture(input_path)
        fps = cap.get(cv2.CAP_PROP_FPS)
        width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
        height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
        total_frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))

        # 输出视频
        output_path = str(Path(input_path).parent / f"{Path(input_path).stem}_output.mp4")
        fourcc = cv2.VideoWriter_fourcc(*'mp4v')
        out = cv2.VideoWriter(output_path, fourcc, fps, (width, height))

        all_detections = []
        frame_count = 0

        while cap.isOpened():
            ret, frame = cap.read()
            if not ret:
                break

            # 检测当前帧
            results = model.predict(frame, conf=conf_threshold, verbose=False)

            # 绘制检测框
            annotated_frame = results[0].plot()
            out.write(annotated_frame)

            # 收集检测结果
            for result in results:
                for box in result.boxes:
                    all_detections.append({
                        "frame": frame_count,
                        "class": result.names[int(box.cls)],
                        "confidence": float(box.conf)
                    })

            frame_count += 1

            # 更新进度
            progress = int((frame_count / total_frames) * 100)
            self.update_state(
                state='PROGRESS',
                meta={'current': frame_count, 'total': total_frames, 'progress': progress}
            )

        cap.release()
        out.release()

        # 更新数据库
        async with AsyncSessionLocal() as db:
            history = await db.get(DetectionHistory, detection_id)
            history.status = DetectionStatus.COMPLETED
            history.output_file_path = output_path
            history.result_json = json.dumps(all_detections)
            history.detected_objects_count = len(all_detections)
            history.completed_at = datetime.utcnow()
            await db.commit()

        return {"status": "completed", "detections": len(all_detections)}

    except Exception as e:
        logger.error(f"视频检测失败: {e}")

        # 更新为失败状态
        async with AsyncSessionLocal() as db:
            history = await db.get(DetectionHistory, detection_id)
            history.status = DetectionStatus.FAILED
            history.error_message = str(e)
            await db.commit()

        raise
```

@router.post("/video")
async def detect_video(
    file: UploadFile = File(...),
    conf_threshold: float = 0.25,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db)

    input_path = upload_dir / f"{file_id}_input{Path(file.filename).suffix}"

    async with aiofiles.open(input_path, 'wb') as f:

    history = DetectionHistory(
        user_id=current_user.id,
        model_id=active_model.id,

        input_file_path=str(input_path),
        status=DetectionStatus.PENDING
    )
    db.add(history)
    await db.commit()
    await db.refresh(history)

    # 提交异步任务
    task = detect_video_task.delay(
        history.id,
        str(input_path),
        active_model.file_path,
        conf_threshold
    )

    history.task_id = task.id
    await db.commit()

    return {
        "detection_id": history.id,
        "task_id": task.id,
        "status": "pending",
        "message": "视频正在处理中，请稍后查询结果"
    }

@router.get("/status/{detection_id}")
async def get_detection_status(
    detection_id: int,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db)
):
    """查询检测任务状态"""
    history = await db.get(DetectionHistory, detection_id)

    if not history or history.user_id != current_user.id:
        raise HTTPException(404, "检测记录不存在")

    # 如果有Celery任务，查询任务进度
    if history.task_id:
        from celery.result import AsyncResult
        task = AsyncResult(history.task_id, app=celery_app)

        if task.state == 'PROGRESS':
            return {
                "status": "processing",
                "progress": task.info.get('progress', 0)
            }

    return {
        "status": history.status,
        "detected_objects_count": history.detected_objects_count,
        "output_url": f"/uploads/detections/{Path(history.output_file_path).name}" if history.output_file_path else None
    }

#### 6.1 创建历史记录API

```python
@router.get("/history", response_model=List[DetectionHistoryResponse])
async def get_detection_history(
    skip: int = 0,
    limit: int = 20,
    input_type: Optional[str] = None,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db)
):
    """获取检测历史记录"""
    query = select(DetectionHistory).where(
        DetectionHistory.user_id == current_user.id
    )

    if input_type:
        query = query.where(DetectionHistory.input_type == input_type)

    query = query.order_by(DetectionHistory.created_at.desc())
    query = query.offset(skip).limit(limit)

    result = await db.execute(query)
    histories = result.scalars().all()

    return histories

@router.delete("/history/{detection_id}")
async def delete_detection_history(
    detection_id: int,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db)
):
    """删除检测记录"""
    history = await db.get(DetectionHistory, detection_id)

    if not history or history.user_id != current_user.id:
        raise HTTPException(404, "检测记录不存在")

    # 删除文件
    if history.input_file_path:
        Path(history.input_file_path).unlink(missing_ok=True)
    if history.output_file_path:
        Path(history.output_file_path).unlink(missing_ok=True)

    await db.delete(history)
    await db.commit()

    return {"message": "删除成功"}
```

## ⚠️ 风险和注意事项

### 技术风险

1. **模型大小**: YOLO模型可能很大（100MB+），影响加载速度
2. **内存占用**: 视频处理可能消耗大量内存
3. **GPU依赖**: 生产环境建议使用GPU加速

### 解决方案

1. 使用轻量级模型（YOLOv8n）
2. 视频分片处理，避免全部加载到内存
3. 提供CPU/GPU两种模式

### 数据安全

1. 用户上传文件需要进行安全检查
2. 限制文件大小（图片10MB，视频100MB）
3. 定期清理过期检测结果
